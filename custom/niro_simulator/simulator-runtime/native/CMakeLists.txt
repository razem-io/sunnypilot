cmake_minimum_required(VERSION 3.20)
project(niro_native_ui LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC OFF)
set(CMAKE_AUTORCC ON)
set(CMAKE_PREFIX_PATH /opt/homebrew/opt/qt@5)
find_package(Qt5 REQUIRED COMPONENTS Widgets Gui Core Network Concurrent DBus Xml Svg)
set(S ${CMAKE_CURRENT_SOURCE_DIR}/../source)
get_filename_component(S ${S} ABSOLUTE)
include_directories(${CMAKE_CURRENT_SOURCE_DIR} ${S} ${S}/msgq ${S}/third_party ${S}/third_party/qrcode /opt/homebrew/include /opt/homebrew/include/eigen3 /opt/homebrew/opt/openssl@3/include)
link_directories(/opt/homebrew/lib /opt/homebrew/opt/openssl@3/lib)
add_compile_definitions(NIRO_DESKTOP_ROAD SUNNYPILOT CL_SILENCE_DEPRECATION GL_SILENCE_DEPRECATION)
add_compile_options(-Wno-deprecated-declarations -UNDEBUG)
include(sources.cmake)
list(FILTER NATIVE_UI_SOURCES EXCLUDE REGEX "/main\\.cc$")
list(APPEND NATIVE_UI_SOURCES ${CMAKE_CURRENT_SOURCE_DIR}/desktop_main.cc)
set(MOC_HEADERS)
foreach(source ${NATIVE_UI_SOURCES})
 string(REGEX REPLACE "\\.cc$" ".h" header ${source})
 if(EXISTS ${header})
  file(READ ${header} body)
  if(body MATCHES "Q_OBJECT")
   list(APPEND MOC_HEADERS ${header})
  endif()
 endif()
endforeach()
qt5_wrap_cpp(MOC_SOURCES ${MOC_HEADERS})
file(GLOB SCHEMA ${S}/cereal/gen/cpp/*.c++)
set(MSG ${S}/msgq_repo/msgq)
add_executable(sunnypilot-ui ${NATIVE_UI_SOURCES} ${MOC_SOURCES} ${SCHEMA} ${S}/selfdrive/assets/assets.qrc
 ${S}/common/params.cc ${S}/common/swaglog.cc ${S}/common/util.cc ${S}/common/watchdog.cc ${S}/common/ratekeeper.cc
 ${S}/common/transformations/orientation.cc ${S}/common/transformations/coordinates.cc
 ${S}/third_party/json11/json11.cpp ${S}/cereal/messaging/socketmaster.cc
 ${MSG}/ipc.cc ${MSG}/event.cc ${MSG}/impl_zmq.cc ${MSG}/impl_msgq.cc ${MSG}/impl_fake.cc ${MSG}/msgq.cc
 ${MSG}/visionipc/visionipc.cc ${MSG}/visionipc/visionipc_server.cc ${MSG}/visionipc/visionipc_client.cc ${MSG}/visionipc/visionbuf.cc ${MSG}/visionipc/visionbuf_cl.cc)
target_link_libraries(sunnypilot-ui Qt5::Widgets Qt5::Gui Qt5::Core Qt5::Network Qt5::Concurrent Qt5::DBus Qt5::Xml Qt5::Svg capnp kj zmq ssl crypto "-framework OpenCL" "-framework OpenGL")
