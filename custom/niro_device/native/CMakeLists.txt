cmake_minimum_required(VERSION 3.16)
project(niro_device_ui LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(S ${NIRO_SOURCE})
get_filename_component(S ${S} ABSOLUTE)
if(NIRO_SDK)
 set(QT_INCLUDE ${NIRO_SDK}/usr/include/aarch64-linux-gnu/qt5)
 set(QT_VERSION 5.12.8)
 add_library(PkgConfig::QT INTERFACE IMPORTED)
 set(QT_INCLUDES ${QT_INCLUDE})
 set(QT_LIBS)
 foreach(component Core Gui Widgets Network Concurrent DBus Xml Svg)
  list(APPEND QT_INCLUDES ${QT_INCLUDE}/Qt${component})
  list(APPEND QT_LIBS ${NIRO_SDK}/usr/lib/aarch64-linux-gnu/libQt5${component}.so)
 endforeach()
 set_target_properties(PkgConfig::QT PROPERTIES INTERFACE_INCLUDE_DIRECTORIES "${QT_INCLUDES}" INTERFACE_LINK_LIBRARIES "${QT_LIBS}")
 include_directories(BEFORE ${NIRO_SDK}/usr/local/include)
 link_directories(${NIRO_SDK}/usr/local/lib)
 add_link_options(-Wl,-rpath-link,${NIRO_SDK}/usr/lib/aarch64-linux-gnu -Wl,-rpath-link,${NIRO_SDK}/lib/aarch64-linux-gnu)
 set(DEVICE_GRAPHICS ${NIRO_SDK}/usr/lib/aarch64-linux-gnu/libOpenCL.so ${NIRO_SDK}/usr/lib/aarch64-linux-gnu/libGLESv2_adreno.so ${NIRO_SDK}/usr/lib/aarch64-linux-gnu/libEGL_adreno.so)
else()
 find_package(PkgConfig REQUIRED)
 pkg_check_modules(QT REQUIRED IMPORTED_TARGET Qt5Widgets Qt5Gui Qt5Core Qt5Network Qt5Concurrent Qt5DBus Qt5Xml Qt5Svg)
 execute_process(COMMAND pkg-config --variable=includedir Qt5Gui OUTPUT_VARIABLE QT_INCLUDE OUTPUT_STRIP_TRAILING_WHITESPACE)
 execute_process(COMMAND pkg-config --modversion Qt5Gui OUTPUT_VARIABLE QT_VERSION OUTPUT_STRIP_TRAILING_WHITESPACE)
 set(DEVICE_GRAPHICS OpenCL GLESv2 EGL)
endif()
include_directories(${QT_INCLUDE}/QtGui/${QT_VERSION}/QtGui ${CMAKE_CURRENT_SOURCE_DIR} ${S} ${S}/msgq ${S}/third_party ${S}/third_party/qrcode ${S}/third_party/OpenCL/include /usr/include/eigen3)
add_compile_definitions(SUNNYPILOT QCOM2 NIRO_NV12_UPLOAD)
add_compile_options(-Wno-deprecated-declarations -UNDEBUG)
include(sources.cmake)
set(MOC_SOURCES)
foreach(source ${NATIVE_UI_SOURCES})
 string(REGEX REPLACE "\\.cc$" ".h" header ${source})
 if(EXISTS ${header})
  file(READ ${header} body)
  if(body MATCHES "Q_OBJECT")
   string(MD5 key ${header})
   set(moc ${CMAKE_CURRENT_BINARY_DIR}/moc_${key}.cpp)
   add_custom_command(OUTPUT ${moc} COMMAND moc -DSUNNYPILOT -DQCOM2 -I${S} -I${S}/msgq ${header} -o ${moc} DEPENDS ${header})
   list(APPEND MOC_SOURCES ${moc})
  endif()
 endif()
endforeach()
set(resource ${CMAKE_CURRENT_BINARY_DIR}/qrc_assets.cpp)
add_custom_command(OUTPUT ${resource} COMMAND rcc ${S}/selfdrive/assets/assets.qrc -o ${resource} DEPENDS ${S}/selfdrive/assets/assets.qrc)
file(GLOB SCHEMA ${S}/cereal/gen/cpp/*.c++)
set(MSG ${S}/msgq_repo/msgq)
add_executable(sunnypilot-ui ${NATIVE_UI_SOURCES} ${MOC_SOURCES} ${SCHEMA} ${resource}
 ${S}/common/params.cc ${S}/common/swaglog.cc ${S}/common/util.cc ${S}/common/watchdog.cc ${S}/common/ratekeeper.cc
 ${S}/common/transformations/orientation.cc ${S}/common/transformations/coordinates.cc
 ${S}/third_party/json11/json11.cpp ${S}/cereal/messaging/socketmaster.cc
 ${MSG}/ipc.cc ${MSG}/event.cc ${MSG}/impl_zmq.cc ${MSG}/impl_msgq.cc ${MSG}/impl_fake.cc ${MSG}/msgq.cc
 ${MSG}/visionipc/visionipc.cc ${MSG}/visionipc/visionipc_server.cc ${MSG}/visionipc/visionipc_client.cc ${MSG}/visionipc/visionbuf.cc ${MSG}/visionipc/visionbuf_cl.cc)
target_link_libraries(sunnypilot-ui PkgConfig::QT capnp kj zmq ssl crypto ${DEVICE_GRAPHICS} pthread wayland-client)
